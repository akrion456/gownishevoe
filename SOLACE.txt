if SERVER then
    -- Singleplayer / Sandbox-first: automatically attach owner HUD on spawn/use
    local function attachOwnerHud(ply, reason)
        if isValid(ply) and ply == owner() then
            local ok, err = pcall(enableHud, ply, true)
            printConsole(string.format("[SOLACE SV] enableHud(%s): ok=%s%s", tostring(reason or "auto"), tostring(ok), ok and "" or (" err=" .. tostring(err))))
            return ok, err
        end
        return false, "not owner"
    end

    hook.add("ClientInitialized", "solace_sv_init_hud", function(ply)
        attachOwnerHud(ply, "ClientInitialized")
    end)

    hook.add("StarfallUsed", "solace_sv_used_hud", function(activator, used)
        if used == chip() then
            local ok, err = attachOwnerHud(activator, "StarfallUsed")
            if isValid(activator) and activator == owner() then
                print("[SOLACE SV] Chip used -> HUD attach=" .. tostring(ok) .. (ok and "" or (" (" .. tostring(err) .. ")")))
            end
        end
    end)

    timer.simple(0.1, function()
        local ok, err = attachOwnerHud(owner(), "startup")
        print("[SOLACE SV] Ready (HUD attach=" .. tostring(ok) .. (ok and "" or (" err=" .. tostring(err))) .. ")")
    end)

    return
end

-- SECTION 3: Core utilities

do (function(SOLACE)
    local Utils = {}
    local _modelExistsCache = {}

    function Utils.sanitizeExportName(name)
        local clean = string.gsub(string.lower(name or "node"), "[^%w_]+", "_")
        clean = string.gsub(clean, "^_+", "")
        clean = string.gsub(clean, "_+$", "")
        if clean == "" or string.match(clean, "^%d") then
            clean = "obj_" .. clean
        end
        return clean
    end

    function Utils.cycleOption(options, current, dir)
        local idx = 1
        for i, v in ipairs(options) do
            if v == current then idx = i break end
        end
        idx = ((idx - 1 + dir) % #options) + 1
        return options[idx]
    end

    function Utils.fmtVec3(v, dec)
        local f = "%." .. (dec or 1) .. "f, %." .. (dec or 1) .. "f, %." .. (dec or 1) .. "f"
        return string.format(f, v[1] or 0, v[2] or 0, v[3] or 0)
    end

    function Utils.formatTimecode(t)
        local mins = math.floor(t / 60)
        local secs = t - mins * 60
        return string.format("%02d:%05.2f", mins, secs)
    end

    function Utils.checkModelMounted(path)
        if not path or path == "" then return false end
        local cached = _modelExistsCache[path]
        if cached ~= nil then return cached end
        local ok = game.modelExists(path)
        _modelExistsCache[path] = ok
        return ok
    end

    SOLACE.Utils = Utils
end)(SOLACE) end

-- ============================================================================
-- SECTION 4: Event system
-- ============================================================================
do (function(SOLACE)

    local Events = {}
    Events.__index = Events

    function Events.new()
        return setmetatable({
            _listeners = {},
        }, Events)
    end

    function Events:on(eventName, id, fn)
        local bucket = self._listeners[eventName]
        if not bucket then
            bucket = {}
            self._listeners[eventName] = bucket
        end
        bucket[id] = fn
    end

    function Events:off(eventName, id)
        local bucket = self._listeners[eventName]
        if bucket then
            bucket[id] = nil
        end
    end

    function Events:emit(eventName, ...)
        local bucket = self._listeners[eventName]
        if not bucket then return end
        for _, fn in pairs(bucket) do
            fn(...)
        end
    end


    SOLACE.Events = Events
end)(SOLACE) end

-- ============================================================================
-- SECTION 5: Command and undo/redo system
-- ============================================================================
do (function(SOLACE)

    local Commands = {}
    Commands.__index = Commands

    function Commands.new(events, maxHistory)
        return setmetatable({
            _events = events,
            _undoStack = {},
            _redoStack = {},
            _maxHistory = maxHistory or 100,
        }, Commands)
    end

    function Commands:execute(cmd)
        if not cmd or type(cmd.redo) ~= "function" or type(cmd.undo) ~= "function" then
            return false
        end

        cmd:redo()
        self._undoStack[#self._undoStack + 1] = cmd
        self._redoStack = {}

        if #self._undoStack > self._maxHistory then
            table.remove(self._undoStack, 1)
        end

        if self._events then
            self._events:emit("historyChanged", self)
        end
        return true
    end

    function Commands:canUndo()
        return #self._undoStack > 0
    end

    function Commands:canRedo()
        return #self._redoStack > 0
    end

    function Commands:getUndoLabel()
        local top = self._undoStack[#self._undoStack]
        return top and top.label or nil
    end

    function Commands:getRedoLabel()
        local top = self._redoStack[#self._redoStack]
        return top and top.label or nil
    end

    function Commands:undo()
        local idx = #self._undoStack
        if idx == 0 then return false end
        local cmd = self._undoStack[idx]
        self._undoStack[idx] = nil
        cmd:undo()
        self._redoStack[#self._redoStack + 1] = cmd
        if self._events then
            self._events:emit("historyChanged", self)
        end
        return true
    end

    function Commands:redo()
        local idx = #self._redoStack
        if idx == 0 then return false end
        local cmd = self._redoStack[idx]
        self._redoStack[idx] = nil
        cmd:redo()
        self._undoStack[#self._undoStack + 1] = cmd
        if self._events then
            self._events:emit("historyChanged", self)
        end
        return true
    end

    function Commands:clear()
        self._undoStack = {}
        self._redoStack = {}
        if self._events then
            self._events:emit("historyChanged", self)
        end
    end


    SOLACE.Commands = Commands
end)(SOLACE) end

-- SECTION 6: Scene IR

do (function(SOLACE)
    local Constants = SOLACE.Constants



    local IR = {}

    local function vec3(x, y, z)
        return { x or 0, y or 0, z or 0 }
    end

    local function ang3(p, y, r)
        return { p or 0, y or 0, r or 0 }
    end

    local function rgba(r, g, b, a)
        return { r or 255, g or 255, b or 255, a or 255 }
    end

    function IR.cloneVec3(v)
        return { v[1] or 0, v[2] or 0, v[3] or 0 }
    end

    function IR.cloneColor(c)
        return { c[1] or 255, c[2] or 255, c[3] or 255, c[4] or 255 }
    end

    -- Runtime Starfall object conversions (used only by adapters/viewport, never stored in IR)
    function IR.toVector(v)
        return Vector(v[1] or 0, v[2] or 0, v[3] or 0)
    end

    function IR.toAngle(a)
        return Angle(a[1] or 0, a[2] or 0, a[3] or 0)
    end

    function IR.toColor(c)
        return Color(c[1] or 255, c[2] or 255, c[3] or 255, c[4] or 255)
    end

    function IR.fromVector(v)
        return { v.x, v.y, v.z }
    end

    function IR.fromAngle(a)
        return { a.p, a.y, a.r }
    end

    function IR.fromColor(c)
        return { c.r, c.g, c.b, c.a }
    end

    function IR.createTransform(opts)
        opts = opts or {}
        return {
            pos   = opts.pos   and IR.cloneVec3(opts.pos)   or vec3(0, 0, 0),
            ang   = opts.ang   and IR.cloneVec3(opts.ang)   or ang3(0, 0, 0),
            scale = opts.scale and IR.cloneVec3(opts.scale) or vec3(1, 1, 1),
            pivot = opts.pivot and IR.cloneVec3(opts.pivot) or vec3(0, 0, 0),
        }
    end

    function IR.createAppearance(opts)
        opts = opts or {}
        return {
            model = opts.model or "",
            material = opts.material or "",
            submaterials = opts.submaterials or {},
            color = opts.color and IR.cloneColor(opts.color) or rgba(255, 255, 255, 255),
            renderMode = opts.renderMode or 0,
            renderFX = opts.renderFX or 0,
            skin = opts.skin or 0,
            bodygroups = opts.bodygroups or {},
            clips = opts.clips or {},
            suppressLighting = opts.suppressLighting or false,
        }
    end

    function IR.createKeyframe(time, value, easing)
        return {
            time = time or 0,
            value = value,
            easing = easing or Constants.EASING.LINEAR,
        }
    end

    function IR.createTrack(propertyPath)
        return {
            property = propertyPath,
            enabled = true,
            keyframes = {},
        }
    end

    function IR.createNode(opts)
        opts = opts or {}
        local kind = opts.kind or Constants.NODE_KINDS.MODEL

        local node = {
            id = opts.id or "",
            name = opts.name or "Node",
            exportName = opts.exportName or "",
            kind = kind,
            parentId = opts.parentId,       -- nil only on root
            children = opts.children or {}, -- ordered array of child node IDs
            visible = opts.visible ~= false,
            locked = opts.locked or false,
            expanded = opts.expanded ~= false,
            description = opts.description or "",
            tags = opts.tags or {},

            transform = IR.createTransform(opts.transform),
            appearance = IR.createAppearance(opts.appearance),

            physics = {
                enabled = false,
                frozen = true,
                mass = 50,
                gravity = true,
                material = "",
            },

            -- Specialized payload blocks (populated for sprite, light, effect, ragdoll, camera)
            sprite = nil,
            light = nil,
            effect = nil,
            bones = nil,
            camera = nil,

            -- Animation tracks keyed by property path (e.g. "transform.pos", "appearance.color")
            tracks = opts.tracks or {},
        }

        if kind == Constants.NODE_KINDS.CAMERA then
            node.camera = {
                fov = (opts.camera and opts.camera.fov) or 80,
                shakeAmplitude = 0,
                shakeFrequency = 0,
                shakeDuration = 0,
            }
        elseif kind == Constants.NODE_KINDS.SPRITE then
            node.sprite = {
                width = 32,
                height = 32,
                roll = 0,
                spawnTime = 0,
                lifetime = 0, -- 0 = persistent
                fadeIn = 0,
                fadeOut = 0,
                additive = false,
            }
        elseif kind == Constants.NODE_KINDS.LIGHT then
            node.light = {
                size = 256,
                brightness = 2,
                decay = 1000,
                projected = false,
                fov = 60,
            }
        end

        return node
    end

    function IR.createProject(opts)
        opts = opts or {}
        local rootId = "root"
        local rootNode = IR.createNode({
            id = rootId,
            name = opts.name or "Untitled Scene",
            kind = Constants.NODE_KINDS.ROOT,
            parentId = nil,
            expanded = true,
            description = "Root container for SOLACE scene hierarchy",
        })

        return {
            schemaVersion = Constants.IR_SCHEMA_VERSION,
            app = Constants.APP_NAME,
            name = opts.name or "Untitled Scene",
            author = opts.author or "",
            createdAt = opts.createdAt or os.time(),
            modifiedAt = opts.modifiedAt or os.time(),
            nextId = 1,
            timeline = {
                fps = 30,
                duration = 10.0,
            },
            origin = { 0, 0, 0 },
            rootId = rootId,
            nodes = {
                [rootId] = rootNode,
            },
            presets = {},
        }
    end


    SOLACE.IR = IR
end)(SOLACE) end

-- SECTION 7: Scene graph
-- ============================================================================
do (function(SOLACE)
    local Constants = SOLACE.Constants
    local Utils = SOLACE.Utils
    local IR = SOLACE.IR



    local SceneGraph = {}
    SceneGraph.__index = SceneGraph

    function SceneGraph.new(events)
        local self = setmetatable({
            _events = events,
            _project = IR.createProject({ name = "Untitled Scene" }),
            _selectedIds = {},
            _selectedMap = {},
            _primarySelectedId = nil,
            _dirty = false,
            _version = 1,
            _flatCacheVersion = -1,
            _flatCacheQuery = nil,
            _flatCache = nil,
            _nodeCountVersion = -1,
            _nodeCountCache = 0,
        }, SceneGraph)

        -- Create initial StudioCamera IR node
        local cam = self:addNode({
            name = "StudioCamera",
            kind = Constants.NODE_KINDS.CAMERA,
            parentId = self._project.rootId,
            description = "Primary editor & cinematic camera actor",
        })
        self:setSelectedId(cam.id)

        return self
    end

    function SceneGraph:getProject()
        return self._project
    end

    function SceneGraph:getRootId()
        return self._project.rootId
    end

    function SceneGraph:getNode(id)
        if not id then return nil end
        return self._project.nodes[id]
    end

    function SceneGraph:getAllNodes()
        return self._project.nodes
    end

    -- Selection API (supports both primary single selection and multi-selection)
    function SceneGraph:getSelectedId()
        return self._primarySelectedId
    end

    function SceneGraph:getSelectedNode()
        return self._primarySelectedId and self._project.nodes[self._primarySelectedId] or nil
    end

    function SceneGraph:getSelectedIds()
        return self._selectedIds
    end

    function SceneGraph:isSelected(id)
        return id ~= nil and self._selectedMap[id] == true
    end

    function SceneGraph:clearSelection()
        if #self._selectedIds == 0 and self._primarySelectedId == nil then return end
        self._selectedIds = {}
        self._selectedMap = {}
        self._primarySelectedId = nil
        if self._events then
            self._events:emit("selectionChanged", nil, nil, self._selectedIds)
        end
    end

    function SceneGraph:setSelectedId(id)
        if id == nil then
            self:clearSelection()
            return
        end
        if not self._project.nodes[id] then return end
        if #self._selectedIds == 1 and self._primarySelectedId == id then return end

        self._selectedIds = { id }
        self._selectedMap = { [id] = true }
        self._primarySelectedId = id
        if self._events then
            self._events:emit("selectionChanged", id, self:getNode(id), self._selectedIds)
        end
    end

    function SceneGraph:toggleSelectedId(id)
        if not id or not self._project.nodes[id] then return end
        if self._selectedMap[id] then
            self._selectedMap[id] = nil
            for i = #self._selectedIds, 1, -1 do
                if self._selectedIds[i] == id then
                    table.remove(self._selectedIds, i)
                    break
                end
            end
            if self._primarySelectedId == id then
                self._primarySelectedId = self._selectedIds[#self._selectedIds]
            end
        else
            self._selectedMap[id] = true
            self._selectedIds[#self._selectedIds + 1] = id
            self._primarySelectedId = id
        end

        if self._events then
            self._events:emit("selectionChanged", self._primarySelectedId, self:getSelectedNode(), self._selectedIds)
        end
    end

    function SceneGraph:isDirty()
        return self._dirty
    end

    function SceneGraph:markDirty()
        self._dirty = true
        self._version = (self._version or 0) + 1
        self._project.modifiedAt = os.time()
        if self._events then
            self._events:emit("sceneDirty", self._project)
        end
    end

    function SceneGraph:bumpHierarchyVersion()
        self._version = (self._version or 0) + 1
    end

    function SceneGraph:generateId(prefix)
        local idx = self._project.nextId
        self._project.nextId = idx + 1
        return string.format("%s_%04d", prefix or "node", idx)
    end

    function SceneGraph:addNode(opts)
        opts = opts or {}
        local parentId = opts.parentId or self._project.rootId
        local parent = self._project.nodes[parentId]
        if not parent then
            parentId = self._project.rootId
            parent = self._project.nodes[parentId]
        end

        local id = opts.id
        if not id or id == "" or self._project.nodes[id] then
            id = self:generateId(opts.kind or "node")
        end

        local node = IR.createNode(opts)
        node.id = id
        node.parentId = parentId
        if node.exportName == "" then
            node.exportName = Utils.sanitizeExportName(node.name) .. "_" .. string.sub(id, -4)
        end

        self._project.nodes[id] = node
        parent.children[#parent.children + 1] = id

        self:markDirty()
        if self._events then
            self._events:emit("nodeAdded", node)
        end
        return node
    end

    function SceneGraph:createModel(name, modelPath, posTbl, angTbl, parentId)
        return self:addNode({
            name = name or "Model",
            kind = Constants.NODE_KINDS.MODEL,
            parentId = parentId or self._project.rootId,
            transform = {
                pos = posTbl or { 0, 0, 0 },
                ang = angTbl or { 0, 0, 0 },
                scale = { 1, 1, 1 },
                pivot = { 0, 0, 0 },
            },
            appearance = {
                model = modelPath,
            },
        })
    end

    function SceneGraph:createGroup(name, posTbl, parentId)
        return self:addNode({
            name = name or "Group",
            kind = Constants.NODE_KINDS.GROUP,
            parentId = parentId or self._project.rootId,
            transform = {
                pos = posTbl or { 0, 0, 0 },
            },
        })
    end

    function SceneGraph:createFolder(name, parentId)
        return self:addNode({
            name = name or "Folder",
            kind = Constants.NODE_KINDS.FOLDER,
            parentId = parentId or self._project.rootId,
        })
    end

    function SceneGraph:createSprite(name, materialPath, posTbl, parentId)
        return self:addNode({
            name = name or "Sprite",
            kind = Constants.NODE_KINDS.SPRITE,
            parentId = parentId or self._project.rootId,
            transform = {
                pos = posTbl or { 0, 0, 0 },
            },
            appearance = {
                material = materialPath or "sprites/light_glow02_add",
            },
        })
    end

    function SceneGraph:removeNode(id)
        if not id or id == self._project.rootId then return false end
        local node = self._project.nodes[id]
        if not node then return false end

        -- Gather entire subtree IDs post-order
        local removedIds = {}
        local function collect(nid)
            local n = self._project.nodes[nid]
            if not n then return end
            for i = 1, #n.children do
                collect(n.children[i])
            end
            removedIds[#removedIds + 1] = nid
        end
        collect(id)

        -- Detach from parent's children array
        local parent = node.parentId and self._project.nodes[node.parentId]
        if parent then
            for i = #parent.children, 1, -1 do
                if parent.children[i] == id then
                    table.remove(parent.children, i)
                    break
                end
            end
        end

        local selChanged = false
        for _, rid in ipairs(removedIds) do
            self._project.nodes[rid] = nil
            if self._selectedMap[rid] then
                self._selectedMap[rid] = nil
                selChanged = true
            end
        end

        if selChanged then
            for i = #self._selectedIds, 1, -1 do
                if not self._selectedMap[self._selectedIds[i]] then
                    table.remove(self._selectedIds, i)
                end
            end
            if not self._selectedMap[self._primarySelectedId] then
                self._primarySelectedId = self._selectedIds[#self._selectedIds]
            end
        end

        self:markDirty()
        if self._events then
            self._events:emit("nodeRemoved", removedIds)
            if selChanged then
                self._events:emit("selectionChanged", self._primarySelectedId, self:getSelectedNode(), self._selectedIds)
            end
        end
        return true
    end

    function SceneGraph:renameNode(id, newName)
        local node = self:getNode(id)
        if not node or not newName then return false end
        local trimmed = string.trim(newName)
        if trimmed == "" then return false end
        node.name = trimmed
        if id == self._project.rootId then
            self._project.name = trimmed
        else
            node.exportName = Utils.sanitizeExportName(trimmed) .. "_" .. string.sub(id, -4)
        end
        self:markDirty()
        if self._events then
            self._events:emit("nodeUpdated", node)
        end
        return true
    end

    function SceneGraph:toggleNodeVisible(id)
        if not id or id == self._project.rootId then return false end
        local node = self:getNode(id)
        if not node then return false end
        node.visible = not node.visible
        self:markDirty()
        if self._events then
            self._events:emit("nodeUpdated", node)
        end
        return true
    end

    function SceneGraph:notifyNodeUpdated(id)
        local node = self:getNode(id)
        if not node then return end
        self:markDirty()
        if self._events then
            self._events:emit("nodeUpdated", node)
        end
    end

    function SceneGraph:getNodeCount()
        if self._nodeCountVersion == self._version then
            return self._nodeCountCache
        end
        local count = 0
        for id in pairs(self._project.nodes) do
            if id ~= self._project.rootId then
                count = count + 1
            end
        end
        self._nodeCountVersion = self._version
        self._nodeCountCache = count
        return count
    end

    -- Flattened depth-first list for Explorer rendering (respects expanded state & optional search filter)
    function SceneGraph:getFlatHierarchy(searchQuery)
        local q = searchQuery and string.lower(string.trim(searchQuery)) or ""
        if self._flatCache and self._flatCacheVersion == self._version and self._flatCacheQuery == q then
            return self._flatCache
        end

        local out = {}
        local hasQuery = q ~= ""

        local function matches(node)
            if not hasQuery then return true end
            return string.find(string.lower(node.name), q, 1, true) ~= nil
                or string.find(string.lower(node.kind), q, 1, true) ~= nil
        end

        local function subtreeMatches(node)
            if matches(node) then return true end
            for _, childId in ipairs(node.children) do
                local child = self._project.nodes[childId]
                if child and subtreeMatches(child) then
                    return true
                end
            end
            return false
        end

        local function walk(nodeId, depth)
            local node = self._project.nodes[nodeId]
            if not node then return end
            if hasQuery and not subtreeMatches(node) then return end

            out[#out + 1] = {
                node = node,
                depth = depth,
                hasChildren = #node.children > 0,
            }

            if node.expanded or hasQuery then
                for _, childId in ipairs(node.children) do
                    walk(childId, depth + 1)
                end
            end
        end

        walk(self._project.rootId, 0)
        self._flatCacheVersion = self._version
        self._flatCacheQuery = q
        self._flatCache = out
        return out
    end

    function SceneGraph:setOrigin(vecTbl)
        self._project.origin = IR.cloneVec3(vecTbl)
        self:markDirty()
    end

    function SceneGraph:setProjectName(name)
        self:renameNode(self._project.rootId, name)
    end


    SOLACE.SceneGraph = SceneGraph
end)(SOLACE) end

-- SECTION 8: Runtime rendering adapter
-- ============================================================================
do (function(SOLACE)
    local Constants = SOLACE.Constants
    local IR = SOLACE.IR



    local SceneRuntime = {}
    SceneRuntime.__index = SceneRuntime

    local DEFAULT_MINS = Vector(-8, -8, -8)
    local DEFAULT_MAXS = Vector(8, 8, 8)

    function SceneRuntime.new(sceneGraph, events)
        local self = setmetatable({
            _scene = sceneGraph,
            _events = events,
            _entries = {},       -- [nodeId] = { kind, holo, material, matPath, worldPos, worldAng, worldScale, boundsMin, boundsMax, visible, errorMsg }
            _dirtyNodes = {},    -- [nodeId] = true
            _pendingRemove = {}, -- array of nodeIds
            _matCache = {},      -- [matPath] = LockedMaterial | false
            _lastError = nil,
        }, SceneRuntime)

        events:on("nodeAdded", "runtime_sync", function(node)
            self:markDirty(node.id)
        end)
        events:on("nodeUpdated", "runtime_sync", function(node)
            self:markSubtreeDirty(node.id)
        end)
        events:on("nodeRemoved", "runtime_sync", function(removedIds)
            for _, id in ipairs(removedIds) do
                self._dirtyNodes[id] = nil
                self._pendingRemove[#self._pendingRemove + 1] = id
            end
        end)

        -- Queue any pre-existing nodes
        for id in pairs(sceneGraph:getAllNodes()) do
            if id ~= sceneGraph:getRootId() then
                self._dirtyNodes[id] = true
            end
        end

        return self
    end

    function SceneRuntime:markDirty(nodeId)
        if nodeId and nodeId ~= self._scene:getRootId() then
            self._dirtyNodes[nodeId] = true
        end
    end

    function SceneRuntime:markSubtreeDirty(nodeId)
        local node = self._scene:getNode(nodeId)
        if not node then return end
        if node.id ~= self._scene:getRootId() then
            self._dirtyNodes[node.id] = true
        end
        for _, childId in ipairs(node.children) do
            self:markSubtreeDirty(childId)
        end
    end

    function SceneRuntime:getLastError()
        return self._lastError
    end

    function SceneRuntime:getEntries()
        return self._entries
    end

    function SceneRuntime:getEntry(nodeId)
        return self._entries[nodeId]
    end

    function SceneRuntime:_loadMaterialCached(matPath)
        if not matPath or matPath == "" then return nil end
        local cached = self._matCache[matPath]
        if cached ~= nil then
            return cached ~= false and cached or nil
        end
        if not hasPermission("material.load") then
            return nil
        end
        local ok, mat = pcall(material.load, matPath)
        if ok and mat then
            self._matCache[matPath] = mat
            return mat
        end
        self._matCache[matPath] = false
        return nil
    end

    function SceneRuntime:_destroyEntry(id)
        local entry = self._entries[id]
        if not entry then return end
        if entry.holo and isValid(entry.holo) then
            pcall(function() entry.holo:remove() end)
        end
        self._entries[id] = nil
    end

    function SceneRuntime:_syncNode(node)
        local id = node.id
        local kind = node.kind
        local entry = self._entries[id]

        local wPos, wAng, wScale = SOLACE.Math3D.computeWorldTransform(self._scene, id)
        local vis = SOLACE.Math3D.isNodeHierarchyVisible(self._scene, id)

        if kind == Constants.NODE_KINDS.MODEL or kind == Constants.NODE_KINDS.HOLOGRAM then
            local modelPath = node.appearance.model or ""
            if modelPath == "" then return end

            if not entry then
                entry = { kind = kind }
                self._entries[id] = entry
            end

            entry.worldPos = wPos
            entry.worldAng = wAng
            entry.worldScale = wScale
            entry.visible = vis

            -- Create or recreate hologram if missing/invalid
            if not entry.holo or not isValid(entry.holo) then
                if not hasPermission("hologram.create") then
                    entry.errorMsg = "Missing hologram.create permission"
                    self._lastError = entry.errorMsg
                    return
                end
                if not hologram.canSpawn() then
                    entry.errorMsg = "Hologram limit/burst reached"
                    self._lastError = entry.errorMsg
                    return
                end
                if not game.modelExists(modelPath) then
                    entry.errorMsg = "Model not found: " .. modelPath
                    self._lastError = entry.errorMsg
                    return
                end

                local ok, holo = pcall(hologram.create, wPos, wAng, modelPath, wScale)
                if not ok or not isValid(holo) then
                    entry.errorMsg = "Failed to spawn hologram: " .. tostring(holo)
                    self._lastError = entry.errorMsg
                    return
                end

                entry.holo = holo
                entry.model = modelPath
                entry.errorMsg = nil
                entry.boundsMin = holo:obbMins() or DEFAULT_MINS
                entry.boundsMax = holo:obbMaxs() or DEFAULT_MAXS
            else
                if entry.model ~= modelPath and game.modelExists(modelPath) then
                    pcall(function() entry.holo:setModel(modelPath) end)
                    entry.model = modelPath
                    entry.boundsMin = entry.holo:obbMins() or DEFAULT_MINS
                    entry.boundsMax = entry.holo:obbMaxs() or DEFAULT_MAXS
                end
                entry.holo:setPos(wPos)
                entry.holo:setAngles(wAng)
                entry.holo:setScale(wScale)
            end

            local holo = entry.holo
            if isValid(holo) then
                holo:setNoDraw(not vis)
                holo:setColor(IR.toColor(node.appearance.color))
                if node.appearance.material and node.appearance.material ~= "" then
                    holo:setMaterial(node.appearance.material)
                end
            end
        elseif kind == Constants.NODE_KINDS.SPRITE then
            if not entry then
                entry = { kind = kind }
                self._entries[id] = entry
            end
            entry.worldPos = wPos
            entry.worldAng = wAng
            entry.worldScale = wScale
            entry.visible = vis
            local spr = node.sprite or { width = 32, height = 32 }
            local hw = (spr.width or 32) * 0.5
            local hh = (spr.height or 32) * 0.5
            entry.boundsMin = Vector(-2, -hw, -hh)
            entry.boundsMax = Vector(2, hw, hh)
            local matPath = node.appearance.material or "sprites/light_glow02_add"
            if entry.matPath ~= matPath then
                entry.matPath = matPath
                entry.material = self:_loadMaterialCached(matPath)
            end
        elseif kind == Constants.NODE_KINDS.GROUP or kind == Constants.NODE_KINDS.FOLDER or kind == Constants.NODE_KINDS.CAMERA then
            if not entry then
                entry = { kind = kind }
                self._entries[id] = entry
            end
            entry.worldPos = wPos
            entry.worldAng = wAng
            entry.worldScale = wScale
            entry.visible = vis
            entry.boundsMin = DEFAULT_MINS
            entry.boundsMax = DEFAULT_MAXS
        end
    end

    -- Must ONLY be called outside render hooks (e.g. inside Think)
    function SceneRuntime:flush()
        if #self._pendingRemove > 0 then
            for i = 1, #self._pendingRemove do
                self:_destroyEntry(self._pendingRemove[i])
            end
            self._pendingRemove = {}
        end

        if next(self._dirtyNodes) ~= nil then
            local batch = {}
            for id in pairs(self._dirtyNodes) do
                batch[#batch + 1] = id
            end
            self._dirtyNodes = {}

            for _, id in ipairs(batch) do
                local node = self._scene:getNode(id)
                if node then
                    self:_syncNode(node)
                else
                    self:_destroyEntry(id)
                end
            end
        end
    end

    -- Returns world transform + scaled bounds for raycasting, outlines & camera focus
    function SceneRuntime:getNodeBounds(nodeId)
        local node = self._scene:getNode(nodeId)
        if not node then return nil end

        local entry = self._entries[nodeId]
        local pos, ang, scale
        if entry and entry.worldPos then
            pos, ang, scale = entry.worldPos, entry.worldAng, entry.worldScale
        else
            pos, ang, scale = SOLACE.Math3D.computeWorldTransform(self._scene, nodeId)
        end

        local rawMin = (entry and entry.boundsMin) or DEFAULT_MINS
        local rawMax = (entry and entry.boundsMax) or DEFAULT_MAXS
        local sMin, sMax = SOLACE.Math3D.getScaledBounds(rawMin, rawMax, scale)
        return pos, ang, sMin, sMax, scale, entry
    end

    -- Draw translucent 3D sprites in PostDrawTranslucentRenderables
    function SceneRuntime:drawSprites(state)
        if not state.studioOpen then return end
        for id, entry in pairs(self._entries) do
            if entry.kind == Constants.NODE_KINDS.SPRITE and entry.visible then
                local node = self._scene:getNode(id)
                if node and entry.worldPos then
                    local spr = node.sprite or { width = 32, height = 32 }
                    local w = (spr.width or 32) * (entry.worldScale and entry.worldScale[1] or 1)
                    local h = (spr.height or 32) * (entry.worldScale and entry.worldScale[2] or 1)
                    local col = IR.toColor(node.appearance.color)

                    if entry.material then
                        render.setMaterial(entry.material)
                        render.draw3DSprite(entry.worldPos, w, h, col)
                    else
                        render.setMaterial(nil)
                        render.setColor(col)
                        render.draw3DSprite(entry.worldPos, w * 0.45, h * 0.45, col)
                    end
                end
            end
        end
        render.setMaterial(nil)
    end

    function SceneRuntime:cleanup()
        for id in pairs(self._entries) do
            self:_destroyEntry(id)
        end
        self._entries = {}
        self._dirtyNodes = {}
        self._pendingRemove = {}
    end


    SOLACE.SceneRuntime = SceneRuntime
end)(SOLACE) end

do (function(SOLACE)

    local Math3D = {}

    -- Compose local matrix: T(pos) * R(ang) * T(pivot) * S(scale) * T(-pivot)
    function Math3D.composeLocalMatrix(t)
        local m = Matrix()
        local pos = t.pos
        local ang = t.ang
        local scale = t.scale
        local pivot = t.pivot

        m:translate(Vector(pos[1] or 0, pos[2] or 0, pos[3] or 0))
        m:rotate(Angle(ang[1] or 0, ang[2] or 0, ang[3] or 0))

        local px, py, pz = pivot[1] or 0, pivot[2] or 0, pivot[3] or 0
        local hasPivot = (px ~= 0 or py ~= 0 or pz ~= 0)
        if hasPivot then
            m:translate(Vector(px, py, pz))
        end

        local sx, sy, sz = scale[1] or 1, scale[2] or 1, scale[3] or 1
        if sx ~= 1 or sy ~= 1 or sz ~= 1 then
            m:scale(Vector(sx, sy, sz))
        end

        if hasPivot then
            m:translate(Vector(-px, -py, -pz))
        end

        return m
    end

    -- Walk parent chain to compute world transform of a Scene IR node.
    -- Returns: worldPos (Vector), worldAng (Angle), worldScale (Vector)
    function Math3D.computeWorldTransform(sceneGraph, nodeId)
        local chain = {}
        local cur = sceneGraph:getNode(nodeId)
        local rootId = sceneGraph:getRootId()
        local guard = 0

        while cur and cur.id ~= rootId and guard < 64 do
            chain[#chain + 1] = cur
            cur = cur.parentId and sceneGraph:getNode(cur.parentId) or nil
            guard = guard + 1
        end

        if #chain == 0 then
            return Vector(0, 0, 0), Angle(0, 0, 0), Vector(1, 1, 1)
        end

        local worldMat = Matrix()
        local sx, sy, sz = 1, 1, 1
        for i = #chain, 1, -1 do
            local n = chain[i]
            local t = n.transform
            -- Compose translation/rotation hierarchy without baking scale into rotation matrix
            local step = Matrix()
            step:translate(Vector(t.pos[1] or 0, t.pos[2] or 0, t.pos[3] or 0))
            step:rotate(Angle(t.ang[1] or 0, t.ang[2] or 0, t.ang[3] or 0))
            worldMat = worldMat * step

            local ns = t.scale
            sx = sx * (ns[1] or 1)
            sy = sy * (ns[2] or 1)
            sz = sz * (ns[3] or 1)
        end

        return worldMat:getTranslation(), worldMat:getAngles(), Vector(sx, sy, sz)
    end

    -- Compute hierarchical visibility (false if any ancestor is hidden)
    function Math3D.isNodeHierarchyVisible(sceneGraph, nodeId)
        local cur = sceneGraph:getNode(nodeId)
        local rootId = sceneGraph:getRootId()
        local guard = 0
        while cur and guard < 64 do
            if cur.visible == false then return false end
            if cur.id == rootId then break end
            cur = cur.parentId and sceneGraph:getNode(cur.parentId) or nil
            guard = guard + 1
        end
        return true
    end

    -- Scale local OBB mins/maxs safely (preserving min <= max even for negative scales)
    function Math3D.getScaledBounds(mins, maxs, scale)
        local sx, sy, sz = scale[1] or 1, scale[2] or 1, scale[3] or 1
        local x1, x2 = (mins[1] or -8) * sx, (maxs[1] or 8) * sx
        local y1, y2 = (mins[2] or -8) * sy, (maxs[2] or 8) * sy
        local z1, z2 = (mins[3] or -8) * sz, (maxs[3] or 8) * sz

        return Vector(math.min(x1, x2), math.min(y1, y2), math.min(z1, z2)),
               Vector(math.max(x1, x2), math.max(y1, y2), math.max(z1, z2))
    end

    -- World center & approximate bounding radius of an OBB
    function Math3D.getOBBCenterAndRadius(pos, ang, sMin, sMax)
        local localCenter = (sMin + sMax) * 0.5
        local halfExt = (sMax - sMin) * 0.5
        local radius = math.max(8, halfExt:getLength())

        local m = Matrix()
        m:translate(pos)
        m:rotate(ang)
        m:translate(localCenter)
        return m:getTranslation(), radius
    end


    SOLACE.Math3D = Math3D
end)(SOLACE) end

do (function(SOLACE)
    local Constants = SOLACE.Constants
    local Math3D = SOLACE.Math3D



    local Freecam = {}
    Freecam.__index = Freecam

    function Freecam.new(state)
        return setmetatable({
            state = state,
            initialized = false,
            camPos = Vector(0, 0, 100),
            camAng = Angle(15, 0, 0),
            targetPos = Vector(0, 0, 100),
            targetAng = Angle(15, 0, 0),
        }, Freecam)
    end

    function Freecam:initFromPlayer()
        local pos = eyePos()
        local ang = eyeAngles()
        -- Offset slightly up & back so user sees the studio origin/chip cleanly
        local initPos = pos + Vector(0, 0, 16)
        self.camPos = initPos:clone()
        self.targetPos = initPos:clone()
        self.camAng = Angle(ang[1], ang[2], 0)
        self.targetAng = Angle(ang[1], ang[2], 0)
        self.initialized = true
    end

    function Freecam:resetToPlayer()
        self:initFromPlayer()
    end

    function Freecam:onMouseMoved(dx, dy)
        if not self.state.studioOpen or not self.state.viewportLooking then return end
        local sens = self.state.camera.sensitivity
        self.targetAng[2] = math.normalizeAngle(self.targetAng[2] - dx * sens)
        self.targetAng[1] = math.clamp(self.targetAng[1] + dy * sens, -89.0, 89.0)
        self.targetAng[3] = 0
    end

    function Freecam:onMouseWheelInViewport(delta)
        if not self.state.studioOpen then return end
        local c = self.state.camera
        local factor = delta > 0 and 1.15 or (1 / 1.15)
        c.speed = math.clamp(math.round(c.speed * factor), Constants.CAMERA_DEFAULTS.minSpeed, Constants.CAMERA_DEFAULTS.maxSpeed)
    end

    function Freecam:focusNode(sceneGraph, runtime, nodeId)
        if not self.initialized then self:initFromPlayer() end
        local targetId = nodeId or sceneGraph:getSelectedId()
        if not targetId or targetId == sceneGraph:getRootId() then return false end

        local node = sceneGraph:getNode(targetId)
        if not node then return false end

        local pos, ang, sMin, sMax = runtime:getNodeBounds(targetId)
        if not pos then return false end

        local center, radius = Math3D.getOBBCenterAndRadius(pos, ang, sMin, sMax)
        local halfFovRad = math.rad(math.clamp(self.state.camera.fov, 30, 120) * 0.5)
        local dist = math.max(55, (radius / math.max(0.15, math.tan(halfFovRad))) * Constants.CAMERA_DEFAULTS.focusPadding)

        local fwd = self.targetAng:getForward()
        self.targetPos = center - fwd * dist
        return true
    end

    function Freecam:getSpawnTransform(distance)
        if not self.initialized then self:initFromPlayer() end
        local d = distance or self.state.spawnDistance or 140
        local p = self.camPos + self.camAng:getForward() * d
        return { math.round(p[1], 2), math.round(p[2], 2), math.round(p[3], 2) },
               { 0, math.round(self.camAng[2], 1), 0 }
    end

    function Freecam:update(dt, ui, vpRect)
        if not self.initialized then
            self:initFromPlayer()
        end

        if not self.state.studioOpen then return end
        if not hasPermission("input") then return end

        -- Adjust speed with mouse wheel when cursor is inside the viewport (or while RMB-looking)
        if ui.wheelDelta ~= 0 and (self.state.viewportLooking or ui:pointInRect(vpRect.x, vpRect.y, vpRect.w, vpRect.h)) then
            self:onMouseWheelInViewport(ui.wheelDelta)
        end

        -- Keyboard movement (disabled while typing into any TextBox)
        if not ui.focusedInputId then
            local fwd = 0
            local right = 0
            local up = 0

            if input.isKeyDown(KEY.W) then fwd = fwd + 1 end
            if input.isKeyDown(KEY.S) then fwd = fwd - 1 end
            if input.isKeyDown(KEY.D) then right = right + 1 end
            if input.isKeyDown(KEY.A) then right = right - 1 end
            if input.isKeyDown(KEY.SPACE) then up = up + 1 end
            if input.isKeyDown(KEY.LCONTROL) or input.isKeyDown(KEY.RCONTROL) then up = up - 1 end

            if fwd ~= 0 or right ~= 0 or up ~= 0 then
                local moveDir = self.targetAng:getForward() * fwd
                    + self.targetAng:getRight() * right
                    + Vector(0, 0, up)

                local len = moveDir:getLength()
                if len > 0.001 then
                    moveDir = moveDir / len
                    local spd = self.state.camera.speed
                    if input.isShiftDown() then
                        spd = spd * self.state.camera.boostMult
                    end
                    self.targetPos = self.targetPos + moveDir * (spd * dt)
                end
            end
        end

        -- Smooth exponential interpolation (frame-rate independent)
        local smooth = self.state.camera.smoothing
        if smooth >= 40 then
            self.camPos = self.targetPos:clone()
            self.camAng = self.targetAng:clone()
        else
            local t = 1 - math.exp(-smooth * dt)
            self.camPos = math.lerpVector(t, self.camPos, self.targetPos)
            self.camAng = math.lerpAngle(t, self.camAng, self.targetAng)
            self.camAng[3] = 0
        end
    end

    function Freecam:calcView(origin, angles, fov, znear, zfar)
        if not self.state.studioOpen then return nil end
        if not self.initialized then
            self:initFromPlayer()
        end
        return {
            origin = self.camPos,
            angles = self.camAng,
            fov = self.state.camera.fov,
            drawviewer = true,
        }
    end

end)(SOLACE) end

do (function(SOLACE)
    local Constants = SOLACE.Constants



    local Selection = {
        hoveredId = nil,
    }

    local RAY_LEN = 16384

    -- Reusable corner scratch vectors
    local p0 = Vector(0, 0, 0)
    local p1 = Vector(0, 0, 0)
    local p2 = Vector(0, 0, 0)
    local p3 = Vector(0, 0, 0)

    local function drawCornerBrackets(pos, ang, sMin, sMax, col, alpha)
        local m = Matrix()
        m:translate(pos)
        m:rotate(ang)

        local x1, y1, z1 = sMin[1], sMin[2], sMin[3]
        local x2, y2, z2 = sMax[1], sMax[2], sMax[3]

        -- Expand slightly (0.5u) to prevent Z-fighting on flat box surfaces
        local pad = 0.5
        x1, y1, z1 = x1 - pad, y1 - pad, z1 - pad
        x2, y2, z2 = x2 + pad, y2 + pad, z2 + pad

        local dx = (x2 - x1) * 0.22
        local dy = (y2 - y1) * 0.22
        local dz = (z2 - z1) * 0.22

        SOLACE.Theme.setColor(col, alpha)

        local xs = { x1, x2 }
        local ys = { y1, y2 }
        local zs = { z1, z2 }

        for ix = 1, 2 do
            local cx = xs[ix]
            local sx = (ix == 1) and dx or -dx
            for iy = 1, 2 do
                local cy = ys[iy]
                local sy = (iy == 1) and dy or -dy
                for iz = 1, 2 do
                    local cz = zs[iz]
                    local sz = (iz == 1) and dz or -dz

                    p0[1], p0[2], p0[3] = cx, cy, cz
                    p1[1], p1[2], p1[3] = cx + sx, cy, cz
                    p2[1], p2[2], p2[3] = cx, cy + sy, cz
                    p3[1], p3[2], p3[3] = cx, cy, cz + sz

                    local w0 = m * p0
                    render.draw3DLine(w0, m * p1, true)
                    render.draw3DLine(w0, m * p2, true)
                    render.draw3DLine(w0, m * p3, true)
                end
            end
        end
    end

    function Selection.pickAtCursor(ui, state, camera, sceneGraph, runtime, vpRect)
        if not state.studioOpen or state.viewportLooking then
            Selection.hoveredId = nil
            return nil
        end
        if not ui:pointInRect(vpRect.x, vpRect.y, vpRect.w, vpRect.h) then
            Selection.hoveredId = nil
            return nil
        end

        local aimDir = trace.aimVector(camera.camAng, state.camera.fov, ui.mx, ui.my, ui.scrW, ui.scrH)
        local rayDelta = aimDir * RAY_LEN

        local bestId = nil
        local bestFrac = 2.0

        for id, entry in pairs(runtime:getEntries()) do
            local node = sceneGraph:getNode(id)
            if node and entry.visible and not node.locked then
                local kind = node.kind
                if kind == Constants.NODE_KINDS.MODEL
                    or kind == Constants.NODE_KINDS.HOLOGRAM
                    or kind == Constants.NODE_KINDS.SPRITE
                then
                    local pos, ang, sMin, sMax = runtime:getNodeBounds(id)
                    if pos then
                        local testAng = (kind == Constants.NODE_KINDS.SPRITE) and camera.camAng or ang
                        local _, _, frac = trace.intersectRayWithOBB(camera.camPos, rayDelta, pos, testAng, sMin, sMax)
                        if frac and frac < bestFrac then
                            bestFrac = frac
                            bestId = id
                        end
                    end
                end
            end
        end

        Selection.hoveredId = bestId
        return bestId
    end

    function Selection.handleClick(ui, state, camera, sceneGraph, runtime, vpRect)
        if not state.studioOpen or state.viewportLooking then return end
        if ui.activeId ~= nil then return end
        if not ui.leftPressed then return end
        if not ui:pointInRect(vpRect.x, vpRect.y, vpRect.w, vpRect.h) then return end

        -- If a HUD widget inside the viewport (none in P2, future gizmos in P3) didn't capture the click:
        local hitId = Selection.pickAtCursor(ui, state, camera, sceneGraph, runtime, vpRect)
        local ctrl = hasPermission("input") and input.isControlDown() or false

        if hitId then
            if ctrl then
                sceneGraph:toggleSelectedId(hitId)
            else
                sceneGraph:setSelectedId(hitId)
            end
        else
            if not ctrl then
                sceneGraph:clearSelection()
            end
        end
    end

    function Selection.draw3D(state, camera, sceneGraph, runtime)
        if not state.studioOpen then return end

        local primaryId = sceneGraph:getSelectedId()
        local selectedIds = sceneGraph:getSelectedIds()

        -- Draw subtle hover box if cursor is over an unselected object
        local hovId = Selection.hoveredId
        if hovId and not sceneGraph:isSelected(hovId) then
            local node = sceneGraph:getNode(hovId)
            if node then
                local pos, ang, sMin, sMax = runtime:getNodeBounds(hovId)
                if pos then
                    local testAng = (node.kind == Constants.NODE_KINDS.SPRITE) and camera.camAng or ang
                    SOLACE.Theme.setColor(SOLACE.Theme.C.textPrimary, 95)
                    render.draw3DWireframeBox(pos, testAng, sMin, sMax, true)
                end
            end
        end

        -- Draw selected object highlights
        for _, id in ipairs(selectedIds) do
            local node = sceneGraph:getNode(id)
            if node and id ~= sceneGraph:getRootId() then
                local pos, ang, sMin, sMax = runtime:getNodeBounds(id)
                if pos then
                    local isPrimary = (id == primaryId)
                    local col = isPrimary and SOLACE.Theme.C.accentBright or SOLACE.Theme.C.accent
                    local testAng = (node.kind == Constants.NODE_KINDS.SPRITE) and camera.camAng or ang

                    -- Subtle wireframe cage + crisp corner brackets
                    SOLACE.Theme.setColor(col, isPrimary and 120 or 80)
                    render.draw3DWireframeBox(pos, testAng, sMin, sMax, true)
                    drawCornerBrackets(pos, testAng, sMin, sMax, col, isPrimary and 255 or 190)
                end
            end
        end
    end


    SOLACE.Selection = Selection
end)(SOLACE) end

    function UIManager:beginFrame(dt)
        self._frameTime = self._frameTime + dt
        self.scrW, self.scrH = render.getResolution()

        self.prevMx, self.prevMy = self.mx, self.my
        if hasPermission("input") then
            local cx, cy = input.getCursorPos()
            self.mx = cx or self.mx
            self.my = cy or self.my
        end
        self.dx = self.mx - self.prevMx
        self.dy = self.my - self.prevMy

        local lDown = hasPermission("input") and input.isMouseDown(MOUSE.LEFT) or false
        local rDown = hasPermission("input") and input.isMouseDown(MOUSE.RIGHT) or false

        self.leftPressed = lDown and not self.leftDown
        self.leftReleased = (not lDown) and self.leftDown
        self.leftDown = lDown

        self.rightPressed = rDown and not self.rightDown
        self.rightReleased = (not rDown) and self.rightDown
        self.rightDown = rDown

        if self.leftPressed then
            -- Clear text focus unless the clicked TextBox re-claims it during draw/update
            self._pendingClearFocus = true
        end

        self.hotId = nil
        self._tooltipText = nil
    end

    function UIManager:endFrame(dt)
        if self._pendingClearFocus then
            self.focusedInputId = nil
            self._pendingClearFocus = false
        end

        if self.leftReleased and self.activeId then
            self.activeId = nil
        end

        if self.hotId and self.hotId == self._lastHoverId and not self.activeId then
            self._tooltipTimer = self._tooltipTimer + dt
        else
            self._tooltipTimer = 0
            self._lastHoverId = self.hotId
        end

        self.wheelDelta = 0
    end

        self.wheelDelta = self.wheelDelta + delta
    end

    function UIManager:pointInRect(x, y, w, h)

    function UIManager:isMouseInClip()
        local top = self._clipStack[#self._clipStack]
        if not top then return true end
        return self.mx >= top[1] and self.mx <= top[3] and self.my >= top[2] and self.my <= top[4]
    end

    function UIManager:hitTest(id, x, y, w, h)
        if self.state.viewportLooking then return false, false end
        local inside = self:pointInRect(x, y, w, h) and self:isMouseInClip()
        local canHover = (self.activeId == nil or self.activeId == id)
        if inside and canHover then
            self.hotId = id
        end
        local isHot = (self.hotId == id)
        local isActive = (self.activeId == id)
        return isHot, isActive
    end

        if self.activeId == id then
            self.activeId = nil
        end
    end

    function UIManager:setTooltip(text)
        if text and text ~= "" then
            self._tooltipText = text
        end
    end

    function UIManager:pushClip(x, y, w, h)
        local x1, y1, x2, y2 = x, y, x + math.max(0, w), y + math.max(0, h)
        local top = self._clipStack[#self._clipStack]
        if top then
            x1 = math.max(x1, top[1])
            y1 = math.max(y1, top[2])
            x2 = math.min(x2, top[3])
            y2 = math.min(y2, top[4])
        end
        self._clipStack[#self._clipStack + 1] = { x1, y1, x2, y2 }
        render.enableScissorRect(x1, y1, math.max(x1, x2), math.max(y1, y2))
    end

    function UIManager:popClip()
        local n = #self._clipStack
        if n == 0 then return end
        self._clipStack[n] = nil
        local top = self._clipStack[n - 1]
        if top then
            render.enableScissorRect(top[1], top[2], math.max(top[1], top[3]), math.max(top[2], top[4]))
        else
            render.disableScissorRect()
        end
    end

    function UIManager:registerTextInput(id, getter, setter, onSubmit)
        self._inputState[id] = {
            get = getter,
            set = setter,
            onSubmit = onSubmit,
        }
    end

    function UIManager:focusTextInput(id)
        self.focusedInputId = id
        self._pendingClearFocus = false
    end

    function UIManager:handleKeyPressed(key)
        if not self.focusedInputId then return false end
        local entry = self._inputState[self.focusedInputId]
        if not entry then
            self.focusedInputId = nil
            return false
        end

        if key == KEY.ESCAPE then
            self.focusedInputId = nil
            return true
        elseif key == KEY.ENTER or key == KEY.PAD_ENTER then
            if entry.onSubmit then entry.onSubmit(entry.get()) end
            self.focusedInputId = nil
            return true
        elseif key == KEY.BACKSPACE then
            local txt = entry.get() or ""
            if #txt > 0 then
                entry.set(string.sub(txt, 1, #txt - 1))
            end
            return true
        end

        local ch = KEY_CHAR_MAP[key]
        if ch then
            local shift = hasPermission("input") and input.isShiftDown() or false
            if shift then
                ch = SHIFT_MAP[ch] or string.upper(ch)
            end
            entry.set((entry.get() or "") .. ch)
            return true
        end

        return false
    end

    function UIManager:shouldBlockBind(bind)
        if not self.state.studioOpen then return false end
        if self.focusedInputId then return true end
        if self.state.viewportLooking then return false end
        return BLOCKED_STUDIO_BINDS[bind] == true
    end

    function UIManager:drawTooltip()
        if not self._tooltipText or self._tooltipTimer < 0.35 or self.activeId then return end

        Theme.setFont(Theme.Fonts.Small)
        local tw, th = render.getTextSize(self._tooltipText)
        local padX, padY = 8, 5
        local bw = tw + padX * 2
        local bh = th + padY * 2

        local tx = math.clamp(self.mx + 14, 6, self.scrW - bw - 6)
        local ty = math.clamp(self.my + 18, 6, self.scrH - bh - 6)

        Theme.setColor(Theme.C.tooltipBg)
        render.drawRectFast(tx, ty, bw, bh)
        Theme.setColor(Theme.C.borderLight)
        render.drawRectOutline(tx, ty, bw, bh, 1)
        Theme.setColor(Theme.C.textPrimary)
        render.drawSimpleText(tx + padX, ty + padY, self._tooltipText, TEXT_ALIGN.LEFT, TEXT_ALIGN.TOP)
    end

    function Widgets.button(ui, id, x, y, w, h, label, opts)
        opts = opts or EMPTY_OPTS
        local disabled = opts.disabled == true
        local active = opts.active == true
        local hot, captured = ui:hitTest(id, x, y, w, h)

        if hot and opts.tooltip then
            ui:setTooltip(opts.tooltip)
        end

        local clicked = false
        if not disabled and hot and ui.leftPressed then
            ui:capture(id)
        end
        if not disabled and captured and ui.leftReleased and hot then
            clicked = true
        end

        -- Background
        if disabled then
            Theme.setColor(Theme.C.bgPanelAlt, 180)
        elseif active then
            Theme.setColor(Theme.C.accentSoft)
        elseif captured and hot then
            Theme.setColor(Theme.C.bgControlPress)
        elseif hot then
            Theme.setColor(Theme.C.bgControlHover)
        else
            Theme.setColor(opts.subtle and Theme.C.bgPanelAlt or Theme.C.bgControl)
        end
        render.drawRectFast(x, y, w, h)

        -- Border (skip outline on idle subtle/disabled buttons to minimize draw calls)
        if active then
            Theme.setColor(Theme.C.accent)
            render.drawRectOutline(x, y, w, h, 1)
        elseif hot and not disabled then
            Theme.setColor(Theme.C.borderLight)
            render.drawRectOutline(x, y, w, h, 1)
        elseif not opts.subtle and not disabled then
            Theme.setColor(Theme.C.border)
            render.drawRectOutline(x, y, w, h, 1)
        end

        local textColor = disabled and Theme.C.textDisabled
            or (active and Theme.C.accentBright or Theme.C.textPrimary)

        local compact = (w < 110)
        local cursorX = x + (compact and 5 or 8)
        if opts.icon then
            local iconSize = math.min(compact and 12 or 14, h - 6)
            local ix = (label == nil or label == "") and (x + (w - iconSize) * 0.5) or cursorX
            local iy = y + (h - iconSize) * 0.5
            Theme.drawIcon(opts.icon, ix, iy, iconSize, textColor)
            cursorX = cursorX + iconSize + (compact and 4 or 6)
        end

        if label and label ~= "" then
            Theme.setFont(opts.font or Theme.Fonts.Body)
            Theme.setColor(textColor)
            if opts.icon then
                render.drawSimpleText(cursorX, y + h * 0.5, label, TEXT_ALIGN.LEFT, TEXT_ALIGN.CENTER)
            else
                render.drawSimpleText(x + w * 0.5, y + h * 0.5, label, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
            end
        end

        if opts.badge and opts.badge ~= "" then
            Theme.setFont(Theme.Fonts.Small)
            local bw = 20
            local bh = 13
            local bx = x + w - bw - 3
            local by = y + 3
            Theme.setColor(Theme.C.bgBadgePhase)
            render.drawRectFast(bx, by, bw, bh)
            Theme.setColor(Theme.C.textMuted)
            render.drawSimpleText(bx + bw * 0.5, by + bh * 0.5, opts.badge, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
        end

        return clicked
    end

    function Widgets.slider(ui, id, x, y, w, h, label, value, minVal, maxVal, step, fmt, tooltip)
        local labelW = math.floor(w * 0.44)
        local trackX = x + labelW
        local trackW = math.max(40, w - labelW)

        Theme.setFont(Theme.Fonts.Body)
        Theme.setColor(Theme.C.textSecondary)
        render.drawSimpleText(x, y + h * 0.5, label, TEXT_ALIGN.LEFT, TEXT_ALIGN.CENTER)

        local hot, captured = ui:hitTest(id, trackX, y, trackW, h)
        if (hot or (ui:pointInRect(x, y, labelW, h) and ui:isMouseInClip())) and tooltip then
            ui:setTooltip(tooltip)
        end

        local newVal = value
        local changed = false
        if hot and ui.leftPressed then
            ui:capture(id)
        end
        if captured and ui.leftDown then
            local frac = math.clamp((ui.mx - trackX) / trackW, 0, 1)
            local raw = minVal + (maxVal - minVal) * frac
            if step and step > 0 then
                raw = math.round(raw / step) * step
            end
            raw = math.clamp(raw, minVal, maxVal)
            if math.abs(raw - value) > 1e-6 then
                newVal = raw
                changed = true
            end
        end

        -- Track background
        Theme.setColor(Theme.C.bgInput)
        render.drawRectFast(trackX, y, trackW, h)

        local frac = math.clamp((newVal - minVal) / math.max(1e-6, maxVal - minVal), 0, 1)
        local fillW = math.floor((trackW - 2) * frac)
        Theme.setColor(captured and Theme.C.accent or (hot and Theme.C.bgSelection or Theme.C.accentSoft), captured and 140 or 180)
        render.drawRectFast(trackX + 1, y + 1, fillW, h - 2)

        if hot or captured then
            Theme.setColor(Theme.C.borderFocus)
            render.drawRectOutline(trackX, y, trackW, h, 1)
        end

        local cache = ui._sliderFmt[id]
        local txt
        if cache and cache.val == newVal and cache.fmt == fmt then
            txt = cache.str
        else
            txt = string.format(fmt or "%.0f", newVal)
            ui._sliderFmt[id] = { val = newVal, fmt = fmt, str = txt }
        end

        Theme.setFont(Theme.Fonts.Mono)
        Theme.setColor(Theme.C.textPrimary)
        render.drawSimpleText(trackX + trackW * 0.5, y + h * 0.5, txt, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)

        return newVal, changed
    end

    function Widgets.textBox(ui, id, x, y, w, h, getter, setter, placeholder, tooltip)

                render.drawLine(x + 25, y + 4, x + 25, y + h - 4)

            if Widgets.button(ui, id .. "_clr", x + w - clearW, y, clearW, h, "x", { subtle = true, tooltip = "Clear search" }) then
                setter("")
            end
        end
    end

    -- 13.4 Resizable Dock Layout & Splitters


    local Layout = {}
    Layout.__index = Layout

    function Layout.new(state)
        local d = Constants.LAYOUT_DEFAULTS
        return setmetatable({
            state = state,
            toolbarH = d.toolbarHeight,
            leftW = d.leftWidth,
            rightW = d.rightWidth,
            bottomH = d.bottomHeight,
            explorerRatio = d.explorerRatio,
            _customized = false,
            _lastScrW = 0,
            _lastScrH = 0,
            rects = {
                toolbar = { x = 0, y = 0, w = 0, h = 0 },
                toolbox = { x = 0, y = 0, w = 0, h = 0 },
                viewport = { x = 0, y = 0, w = 0, h = 0 },
                explorer = { x = 0, y = 0, w = 0, h = 0 },
                properties = { x = 0, y = 0, w = 0, h = 0 },
                timeline = { x = 0, y = 0, w = 0, h = 0 },
            },
            _splitters = {},
        }, Layout)
    end

    function Layout:applyResponsiveDefaults(scrW, scrH)
        local d = Constants.LAYOUT_DEFAULTS
        self.toolbarH = (scrH < 820) and 34 or d.toolbarHeight
        self.leftW = math.clamp(math.floor(scrW * 0.145), d.minLeftWidth, d.maxLeftWidth)
        if self.leftW > 244 then self.leftW = 244 end
        self.rightW = math.clamp(math.floor(scrW * 0.175), d.minRightWidth, d.maxRightWidth)
        if self.rightW > 288 then self.rightW = 288 end
        self.bottomH = math.clamp(math.floor(scrH * 0.18), d.minBottomHeight, d.maxBottomHeight)
        if self.bottomH > 184 then self.bottomH = 184 end
        self.explorerRatio = d.explorerRatio
    end

    function Layout:reset()
        self._customized = false
        if self._lastScrW > 0 and self._lastScrH > 0 then
            self:applyResponsiveDefaults(self._lastScrW, self._lastScrH)
        else
            local d = Constants.LAYOUT_DEFAULTS
            self.toolbarH = d.toolbarHeight
            self.leftW = d.leftWidth
            self.rightW = d.rightWidth
            self.bottomH = d.bottomHeight
            self.explorerRatio = d.explorerRatio
        end
    end

    function Layout:compute(scrW, scrH)
        local d = Constants.LAYOUT_DEFAULTS
        if scrW ~= self._lastScrW or scrH ~= self._lastScrH then
            self._lastScrW = scrW
            self._lastScrH = scrH
            if not self._customized then
                self:applyResponsiveDefaults(scrW, scrH)
            end
        end
        local sp = d.splitterSize
        local panels = self.state.panels

        local showLeft = panels.toolbox
        local showRight = panels.explorer or panels.properties
        local showBottom = panels.timeline

        local topY = self.toolbarH
        local availH = math.max(200, scrH - topY)

        -- Clamp bottom timeline height
        local activeBottomH = 0
        if showBottom then
            local maxB = math.min(d.maxBottomHeight, availH - d.minViewportHeight - sp)
            self.bottomH = math.clamp(self.bottomH, d.minBottomHeight, math.max(d.minBottomHeight, maxB))
            activeBottomH = self.bottomH
        end

        local middleH = availH - (showBottom and (activeBottomH + sp) or 0)

        -- Clamp left & right widths while preserving minimum center viewport width
        local activeLeftW = showLeft and math.clamp(self.leftW, d.minLeftWidth, d.maxLeftWidth) or 0
        local activeRightW = showRight and math.clamp(self.rightW, d.minRightWidth, d.maxRightWidth) or 0

        local neededNonViewport = (showLeft and (activeLeftW + sp) or 0) + (showRight and (activeRightW + sp) or 0)
        if scrW - neededNonViewport < d.minViewportWidth then
            local maxCombined = math.max(0, scrW - d.minViewportWidth - (showLeft and sp or 0) - (showRight and sp or 0))
            if showLeft and showRight then
                local ratio = activeLeftW / math.max(1, activeLeftW + activeRightW)
                activeLeftW = math.floor(maxCombined * ratio)
                activeRightW = maxCombined - activeLeftW
            elseif showLeft then
                activeLeftW = maxCombined
            elseif showRight then
                activeRightW = maxCombined
            end
        end

        self.leftW = showLeft and activeLeftW or self.leftW
        self.rightW = showRight and activeRightW or self.rightW

        local r = self.rects
        r.toolbar = { x = 0, y = 0, w = scrW, h = self.toolbarH }

        local curX = 0
        if showLeft then
            r.toolbox = { x = 0, y = topY, w = activeLeftW, h = middleH }
            curX = activeLeftW + sp
        else
            r.toolbox = { x = 0, y = 0, w = 0, h = 0 }
        end

        local vpW = scrW - curX - (showRight and (activeRightW + sp) or 0)
        r.viewport = { x = curX, y = topY, w = math.max(0, vpW), h = middleH }

        local rightX = curX + r.viewport.w + (showRight and sp or 0)
        if panels.explorer and panels.properties then
            local expH = math.floor((middleH - sp) * math.clamp(self.explorerRatio, d.minExplorerRatio, d.maxExplorerRatio))
            local propY = topY + expH + sp
            r.explorer = { x = rightX, y = topY, w = activeRightW, h = expH }
            r.properties = { x = rightX, y = propY, w = activeRightW, h = math.max(0, middleH - expH - sp) }
        elseif panels.explorer then
            r.explorer = { x = rightX, y = topY, w = activeRightW, h = middleH }
            r.properties = { x = 0, y = 0, w = 0, h = 0 }
        elseif panels.properties then
            r.explorer = { x = 0, y = 0, w = 0, h = 0 }
            r.properties = { x = rightX, y = topY, w = activeRightW, h = middleH }
        else
            r.explorer = { x = 0, y = 0, w = 0, h = 0 }
            r.properties = { x = 0, y = 0, w = 0, h = 0 }
        end

        if showBottom then
            r.timeline = { x = 0, y = topY + middleH + sp, w = scrW, h = activeBottomH }
        else
            r.timeline = { x = 0, y = 0, w = 0, h = 0 }
        end

        -- Build active splitter hitboxes
        self._splitters = {}
        if showLeft then
            self._splitters[#self._splitters + 1] = {
                id = "split_left",
                axis = "x",
                x = r.toolbox.w,
                y = topY,
                w = sp,
                h = middleH,
            }
        end
        if showRight then
            self._splitters[#self._splitters + 1] = {
                id = "split_right",
                axis = "x",
                x = r.viewport.x + r.viewport.w,
                y = topY,
                w = sp,
                h = middleH,
            }
        end
        if panels.explorer and panels.properties then
            self._splitters[#self._splitters + 1] = {
                id = "split_right_y",
                axis = "y",
                x = rightX,
                y = r.explorer.y + r.explorer.h,
                w = activeRightW,
                h = sp,
                parentTop = topY,
                parentH = middleH - sp,
            }
        end
        if showBottom then
            self._splitters[#self._splitters + 1] = {
                id = "split_bottom",
                axis = "y",
                x = 0,
                y = topY + middleH,
                w = scrW,
                h = sp,
            }
        end
    end

    function Layout:updateAndDrawSplitters(ui)
        local d = Constants.LAYOUT_DEFAULTS
        for _, s in ipairs(self._splitters) do
            local hot, captured = ui:hitTest(s.id, s.x - 1, s.y - 1, s.w + 2, s.h + 2)
            if hot then
                ui:setTooltip("Drag to resize panel")
            end
            if hot and ui.leftPressed then
                ui:capture(s.id)
            end

            if captured and ui.leftDown then
                self._customized = true
                if s.id == "split_left" then
                    self.leftW = math.clamp(ui.mx, d.minLeftWidth, d.maxLeftWidth)
                elseif s.id == "split_right" then

    local function cycleFps(current, dir)
        local opts = Constants.FPS_OPTIONS
        local idx = 2
        for i, v in ipairs(opts) do
            if v == current then idx = i break end
        end
        idx = ((idx - 1 + dir) % #opts) + 1
        return opts[idx]
    end

    function Timeline.draw(ui, state, sceneGraph, rect)
        if rect.w <= 0 or rect.h <= 0 then return end

        local tl = state.timeline
        local frameIdx = math.floor(tl.currentTime * tl.fps + 0.5)
        local totalFrames = math.floor(tl.duration * tl.fps + 0.5)
        if frameIdx ~= Timeline._lastFrame or totalFrames ~= Timeline._lastTotal or tl.fps ~= Timeline._lastFps then
            Timeline._lastFrame = frameIdx
            Timeline._lastTotal = totalFrames
            Timeline._lastFps = tl.fps
            Timeline._badgeStr = string.format("%s / %s  |  F %d / %d (%d FPS)",
                Utils.formatTimecode(tl.currentTime),
                Utils.formatTimecode(tl.duration),
                frameIdx,
                totalFrames,
                tl.fps

        if ui.wheelDelta ~= 0 and ui:pointInRect(rulerX, by, rulerW, bh) and ui:isMouseInClip() then
            local factor = ui.wheelDelta > 0 and 1.15 or 0.87
            tl.zoom = math.clamp(tl.zoom * factor, 25, 320)
        end

        -- Ruler scrubbing (click or drag anywhere on ruler)
        local rulerHot, rulerCap = ui:hitTest("tl_ruler", rulerX, rulerY, rulerW, stripH)
        if rulerHot then
            ui:setTooltip("Click or drag to scrub playhead (Mouse wheel to zoom)")
            if ui.leftPressed then
                ui:capture("tl_ruler")
            end
        end
        if rulerCap and ui.leftDown then
            local t = tl.scrollTime + (ui.mx - rulerX - 10) / tl.zoom
            tl.currentTime = math.clamp(t, 0, tl.duration)
        end

        ui:pushClip(rulerX, by, rulerW, bh)

        -- Ruler background & ticks
        Theme.setColor(Theme.C.bgInput)
        render.drawRectFast(rulerX, rulerY, rulerW, stripH - 1)

        local originPX = rulerX + 10 - tl.scrollTime * tl.zoom
        local durPX = originPX + tl.duration * tl.zoom

        -- Active sequence span highlight
        local spanX1 = math.max(rulerX, originPX)
        local spanX2 = math.min(rulerX + rulerW, durPX)
        if spanX2 > spanX1 then
            Theme.setColor(Theme.C.bgPanel, 220)
            render.drawRectFast(spanX1, trackAreaY, spanX2 - spanX1, trackAreaH)
        end

        local secStep = tl.zoom < 50 and 2 or 1
        local subDiv = tl.zoom >= 110 and 4 or (tl.zoom >= 65 and 2 or 1)
        local minSec = math.max(0, math.floor((rulerX - 20 - originPX) / tl.zoom))
        local maxSec = math.min(math.ceil(tl.duration), math.ceil((rulerX + rulerW + 20 - originPX) / tl.zoom))

        Theme.setFont(Theme.Fonts.Small)
        for sec = minSec, maxSec do
            local sx = originPX + sec * tl.zoom
            if sec % secStep == 0 then
                Theme.setColor(Theme.C.borderLight)
                render.drawLine(sx, rulerY + 12, sx, rulerY + stripH - 2)
                Theme.setColor(Theme.C.textSecondary)
                render.drawSimpleText(sx + 3, rulerY + 4, SEC_LABELS[sec] or (sec .. "s"), TEXT_ALIGN.LEFT, TEXT_ALIGN.TOP)

                -- Vertical guide line into track body
                Theme.setColor(Theme.C.border, 110)
                render.drawLine(sx, trackAreaY, sx, trackAreaY + trackAreaH)
            else
                Theme.setColor(Theme.C.border)
                render.drawLine(sx, rulerY + 18, sx, rulerY + stripH - 2)
            end

            if subDiv > 1 and sec < tl.duration then
                for sub = 1, subDiv - 1 do
                    local subX = sx + (sub / subDiv) * tl.zoom
                    Theme.setColor(Theme.C.border, 140)
                    render.drawLine(subX, rulerY + 21, subX, rulerY + stripH - 2)
                end
            end
        end

        -- End-of-sequence marker
        if durPX >= rulerX and durPX <= rulerX + rulerW then
            Theme.setColor(Theme.C.warning, 160)
            render.drawLine(durPX, rulerY, durPX, by + bh)
        end

        -- Playhead scrubber line + handle
        local playX = math.floor(originPX + tl.currentTime * tl.zoom)
        if playX >= rulerX and playX <= rulerX + rulerW then
            Theme.setColor(Theme.C.accent)
            render.drawLine(playX, rulerY + 4, playX, by + bh)
            render.drawRectFast(playX - 5, rulerY + 3, 11, 10)
            Theme.setColor(Theme.C.textPrimary)
            render.drawRectOutline(playX - 5, rulerY + 3, 11, 10, 1)
        end

        -- Track rows preview (mirrors current IR nodes)
        local selId = sceneGraph:getSelectedId()
        local rows = sceneGraph:getFlatHierarchy("")
        local rowY = trackAreaY + 4
        local rowH = 22
        for _, item in ipairs(rows) do
            if item.node.id ~= sceneGraph:getRootId() and rowY + rowH <= by + bh - 4 then
                local isSel = (item.node.id == selId)
                if isSel then
                    Theme.setColor(Theme.C.accentSoft, 80)
                    render.drawRectFast(rulerX, rowY, rulerW, rowH)
                end
                Theme.setColor(Theme.C.border, 90)
                render.drawLine(rulerX, rowY + rowH, rulerX + rulerW, rowY + rowH)
                rowY = rowY + rowH
            end
        end

        if trackAreaH >= 48 then
            Theme.setFont(Theme.Fonts.Small)
            Theme.setColor(Theme.C.textMuted)
            render.drawSimpleText(
                rulerX + 12,
                by + bh - 14,
                "Keyframe tracks, interpolation & easing curves arrive in Phase 8 (Ruler scrubbing is live).",
                TEXT_ALIGN.LEFT,
                TEXT_ALIGN.CENTER
            )
        end

        ui:popClip()

        -- Left track label list (clipped to headerColW)
        ui:pushClip(bx, trackAreaY, headerColW, trackAreaH)
        local lRowY = trackAreaY + 4
        for _, item in ipairs(rows) do
            if item.node.id ~= sceneGraph:getRootId() and lRowY + rowH <= by + bh - 4 then
                local isSel = (item.node.id == selId)
                if isSel then
                    Theme.setColor(Theme.C.bgSelectionRow)
                    render.drawRectFast(bx + 2, lRowY, headerColW - 4, rowH)
                end
                Theme.setFont(Theme.Fonts.Body)
                Theme.setColor(isSel and Theme.C.textPrimary or Theme.C.textSecondary)
                render.drawSimpleText(bx + 10, lRowY + rowH * 0.5, item.node.name, TEXT_ALIGN.LEFT, TEXT_ALIGN.CENTER)

                Theme.setColor(Theme.C.border, 90)
                render.drawLine(bx, lRowY + rowH, bx + headerColW, lRowY + rowH)
                lRowY = lRowY + rowH
            end
        end
        ui:popClip()

        ui:popClip()
    end

    SOLACE.Timeline = Timeline
end)(SOLACE) end

    local rtSupported = (type(render.createRenderTarget) == "function")
        and (type(render.selectRenderTarget) == "function")
        and (type(render.setRenderTargetTexture) == "function")
        and (type(render.drawTexturedRectUV) == "function")

    local panelCaches = {
        toolbar    = { rtName = "sol_rt_tb",   w = 0, h = 0, sig = nil, wasInPanel = false, wasActive = false, created = false, failed = false, localRect = { x = 0, y = 0, w = 0, h = 0 } },
        toolbox    = { rtName = "sol_rt_tbx",  w = 0, h = 0, sig = nil, wasInPanel = false, wasActive = false, created = false, failed = false, localRect = { x = 0, y = 0, w = 0, h = 0 } },
        explorer   = { rtName = "sol_rt_exp",  w = 0, h = 0, sig = nil, wasInPanel = false, wasActive = false, created = false, failed = false, localRect = { x = 0, y = 0, w = 0, h = 0 } },
        properties = { rtName = "sol_rt_prop", w = 0, h = 0, sig = nil, wasInPanel = false, wasActive = false, created = false, failed = false, localRect = { x = 0, y = 0, w = 0, h = 0 } },
        timeline   = { rtName = "sol_rt_tl",   w = 0, h = 0, sig = nil, wasInPanel = false, wasActive = false, created = false, failed = false, localRect = { x = 0, y = 0, w = 0, h = 0 } },
    }

    local function drawCachedPanel(key, rect, sig, isActiveInPanel, drawFn)
        local w, h = rect.w, rect.h
        if w <= 0 or h <= 0 then return end

        local c = panelCaches[key]
        local numStrips = math.ceil(w / 1024)
        local totalRTH = numStrips * (h + 2) - 2
        if not rtSupported or c.failed or h > 1024 or totalRTH > 1024 then
            drawFn(rect)
            return
        end

        if not c.created then
            local ok = pcall(render.createRenderTarget, c.rtName)
            if not ok then
                c.failed = true
                drawFn(rect)
                return
            end
            c.created = true
        end

        local inPanel = ui:pointInRect(rect.x, rect.y, w, h)
        local dirty = (w ~= c.w) or (h ~= c.h) or (sig ~= c.sig)
            or inPanel or c.wasInPanel
            or isActiveInPanel or c.wasActive
            or (ui.wheelDelta ~= 0 and inPanel)

        if dirty then
            local realMx, realMy = ui.mx, ui.my
            render.selectRenderTarget(c.rtName)
            Theme.beginFrame()
            if numStrips == 1 then
                ui.mx = realMx - rect.x
                ui.my = realMy - rect.y
                c.localRect.x = 0
                c.localRect.y = 0
                c.localRect.w = w
                c.localRect.h = h
                drawFn(c.localRect)
            else
                for s = 0, numStrips - 1 do
                    local offX = s * 1024
                    local offY = s * (h + 2)
                    local stripW = math.min(1024, w - offX)
                    ui.mx = realMx - rect.x - offX
                    ui.my = realMy - rect.y - offY
                    c.localRect.x = -offX
                    c.localRect.y = offY
                    c.localRect.w = w
                    c.localRect.h = h
                    ui:pushClip(0, offY, stripW, h)
                    drawFn(c.localRect)
                    ui:popClip()
                end
            end
            ui.mx = realMx
            ui.my = realMy
            render.selectRenderTarget(nil)
            Theme.beginFrame()

            c.w = w
            c.h = h
            c.sig = sig
            c.wasInPanel = inPanel
            c.wasActive = isActiveInPanel
        end

        render.setRenderTargetTexture(c.rtName)
        Theme.setRGBA(255, 255, 255, 255)
        if numStrips == 1 then
            render.drawTexturedRectUV(rect.x, rect.y, w, h, 0, 0, w / 1024, h / 1024)
        else
            for s = 0, numStrips - 1 do
                local offX = s * 1024
                local offY = s * (h + 2)
                local stripW = math.min(1024, w - offX)
                render.drawTexturedRectUV(
                    rect.x + offX, rect.y, stripW, h,
                    0, offY / 1024, stripW / 1024, (offY + h) / 1024
                )
            end
        end
        render.setRenderTargetTexture(nil)
    end

    hook.add("DrawHUD", "solace_hud_paint", function()
        if not hasPermission("render.hud") then return end
        hudConnected = true

        local now = timer.realtime()
        local dt = math.clamp(now - lastFrameTime, 0.0001, 0.25)
        lastFrameTime = now
        local instFps = 1 / dt
        fpsSmooth = fpsSmooth * 0.92 + instFps * 0.08

        if not firstDrawLogged then
            firstDrawLogged = true
            local sw, sh = render.getResolution()
            printConsole(string.format("[SOLACE CL] DrawHUD active (%dx%d, studioOpen=%s)", sw or 0, sh or 0, tostring(state.studioOpen)))
        end

        Theme.beginFrame()

        if not state.studioOpen then
            local scrW = render.getResolution()
            local label = "SOLACE Minimized  |  Press F6 to Open"
            Theme.setFont(Theme.Fonts.Small)
            local tw = render.getTextSize(label)
            local pw = tw + 18
            local px = scrW - pw - 14
            local py = 14
            Theme.setColor(Theme.C.hudPill)
            render.drawRectFast(px, py, pw, 24)
            Theme.setColor(Theme.C.borderLight)
            render.drawRectOutline(px, py, pw, 24, 1)
            Theme.setColor(Theme.C.textPrimary)
            render.drawSimpleText(px + 9, py + 12, label, TEXT_ALIGN.LEFT, TEXT_ALIGN.CENTER)
            return
        end

        local ok, err = pcall(function()
            syncCursor()
            ui:beginFrame(dt)
            layout:compute(ui.scrW, ui.scrH)

            local rects = layout.rects
            ViewportGrid.updateLookMode(state, ui, rects.viewport)
            camera:update(dt, ui, rects.viewport)

            local sel = sceneGraph:getSelectedNode()
            local selIds = sceneGraph:getSelectedIds()
            actions.hasSelection = (sel ~= nil and sel.id ~= sceneGraph:getRootId())

            local actId = ui.activeId
            local focId = ui.focusedInputId
            local blinkPhase = math.floor(ui._frameTime * 2)
            local actPrefix3 = actId and string.sub(actId, 1, 3) or ""
            local actPrefix4 = actId and string.sub(actId, 1, 4) or ""

            -- Draw docked studio panels around the live center 3D viewport cutout (1:1 RT cached when idle)
            local tbSig = state.activeTool .. ":" .. tostring(actions.hasSelection)
                .. ":" .. tostring(state.grid.enabled) .. tostring(state.grid.showAxes) .. tostring(state.grid.showTelemetry)
                .. ":" .. tostring(state.panels.toolbox) .. tostring(state.panels.explorer) .. tostring(state.panels.properties) .. tostring(state.panels.timeline)
            drawCachedPanel("toolbar", rects.toolbar, tbSig, actPrefix3 == "tb_", function(r)
                Toolbar.draw(ui, state, layout, r, actions)
            end)

            local tbxFocus = (focId == "toolbox_search" or focId == "tbx_custom_mdl")
            local tbxSig = state.activeToolboxCategory .. "|" .. state.toolboxSearch .. "|" .. state.customModelPath
                .. "|" .. tostring(Toolbox._scrollY) .. "|" .. (tbxFocus and (focId .. ":" .. blinkPhase) or "")
            drawCachedPanel("toolbox", rects.toolbox, tbxSig, actPrefix4 == "tbx_" or actPrefix4 == "tool", function(r)
                Toolbox.draw(ui, state, r, actions)
            end)

            local expFocus = (focId == "explorer_search")
            local expSig = sceneGraph._version .. "|" .. tostring(sel and sel.id) .. "|" .. #selIds
                .. "|" .. state.explorerSearch .. "|" .. tostring(Explorer._scrollY) .. "|" .. (expFocus and blinkPhase or "")
            drawCachedPanel("explorer", rects.explorer, expSig, actPrefix4 == "exp_" or actPrefix4 == "expl", function(r)
                Explorer.draw(ui, state, sceneGraph, r, actions)
            end)

            local propFocus = (focId ~= nil and string.sub(focId, 1, 15) == "prop_node_name_")
            local selXform = sel and (sel.transform.pos[1] .. "," .. sel.transform.pos[2] .. "," .. sel.transform.pos[3]
                .. "," .. sel.transform.ang[1] .. "," .. sel.transform.ang[2] .. "," .. sel.transform.ang[3]
                .. "," .. sel.transform.scale[1] .. "," .. sel.transform.scale[2] .. "," .. sel.transform.scale[3]) or ""
            local c = state.camera
            local g = state.grid
            local s = state.sections
            local propSig = sceneGraph._version .. "|" .. tostring(sel and sel.id) .. "|" .. tostring(sel and sel.name)
                .. "|" .. tostring(sel and sel.visible) .. "|" .. #selIds .. "|" .. selXform
                .. "|" .. c.speed .. ":" .. c.boostMult .. ":" .. c.sensitivity .. ":" .. c.fov .. ":" .. c.smoothing
                .. "|" .. tostring(g.enabled) .. ":" .. g.spacing .. ":" .. g.halfExtents .. ":" .. g.originMode
                .. ":" .. tostring(g.showOrientationGizmo) .. ":" .. tostring(g.showTelemetry)
                .. "|" .. tostring(s.selection) .. tostring(s.camera) .. tostring(s.grid) .. tostring(s.origin) .. tostring(s.quotas)
                .. "|" .. (s.quotas and math.floor(ui._frameTime * 5) or 0)
                .. "|" .. (propFocus and blinkPhase or "")
                .. "|" .. tostring(Properties._scrollY)
            drawCachedPanel("properties", rects.properties, propSig, actPrefix4 == "prop" or actPrefix4 == "sec_", function(r)
                Properties.draw(ui, state, sceneGraph, runtime, camera, r, actions)
            end)

            local tl = state.timeline
            local tlSig = sceneGraph._version .. "|" .. tostring(sel and sel.id)
                .. "|" .. tl.currentTime .. "|" .. tl.duration .. "|" .. tl.fps .. "|" .. tl.zoom .. "|" .. tl.scrollTime
            drawCachedPanel("timeline", rects.timeline, tlSig, actPrefix3 == "tl_", function(r)
                Timeline.draw(ui, state, sceneGraph, r)
            end)

            -- Draw viewport frame, 3D orientation triad & telemetry
            ViewportGrid.drawOverlay(state, ui, rects.viewport, fpsSmooth, camera, sceneGraph)

            -- Draw interactive panel splitters on top of panel borders
            layout:updateAndDrawSplitters(ui)

            -- Viewport raycast hover & click selection (only when no UI widget/splitter is hovered/active)
            if ui.hotId == nil and ui.activeId == nil then
                Selection.pickAtCursor(ui, state, camera, sceneGraph, runtime, rects.viewport)
                Selection.handleClick(ui, state, camera, sceneGraph, runtime, rects.viewport)
            else
                Selection.hoveredId = nil
            end
        end)

        if not ok then
            if render.isInRenderTarget and render.isInRenderTarget() then
                pcall(render.selectRenderTarget, nil)
            end
            if render.setRenderTargetTexture then
                pcall(render.setRenderTargetTexture, nil)
            end
            while #ui._clipStack > 0 do
                ui:popClip()
            end
            if not drawErrorLogged then
                drawErrorLogged = true
                print("[SOLACE] DrawHUD error: " .. tostring(err))
                printConsole("[SOLACE] DrawHUD error: " .. tostring(err))
            end
            local sw = ui.scrW or render.getResolution()
            render.setFont(Theme.Fonts.BodyBold)
            Theme.setColor(Theme.C.danger, 230)
            render.drawRectFast(12, 12, math.min(sw - 24, 720), 34)
            Theme.setColor(Theme.C.textPrimary)
            render.drawSimpleText(22, 29, "SOLACE Render Error: " .. tostring(err), TEXT_ALIGN.LEFT, TEXT_ALIGN.CENTER)
        end

        -- Permission warning banner if cursor/input/calcview permissions were restricted
        if not hasPermission("input.enablecursor") or not hasPermission("input") or not hasPermission("render.calcview") then
            local rects = layout.rects
            local sw = ui.scrW or render.getResolution()
            local msg = "SOLACE needs Cursor/Input/CalcView permissions: Press E (Use) on the Starfall chip to grant."
            render.setFont(Theme.Fonts.BodyBold)
            local mw = render.getTextSize(msg) + 24
            local mx = math.floor((sw - mw) * 0.5)
            local my = (rects and rects.toolbar and rects.toolbar.h or 36) + 12
            Theme.setColor(Theme.C.danger, 225)
            render.drawRectFast(mx, my, mw, 28)
            Theme.setColor(Theme.C.textPrimary)
            render.drawSimpleText(mx + mw * 0.5, my + 14, msg, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
        end

        ui:drawTooltip()
        ui:endFrame(dt)
    end)
end)(SOLACE) end
